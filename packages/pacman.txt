# Explicitly-installed packages from the official repos.
#
# Install (the sed strips these comments -- pacman can't read them):
#   sed 's/#.*//' packages/pacman.txt | sudo pacman -S --needed -
#
# Sections run most- to least- essential. Only "Hyprland desktop" and "Fonts"
# are genuinely required for this config to work. Delete any section you don't
# want -- no later section depends on an earlier one.

# --- Base system -----------------------------------------------------------
# MACHINE SPECIFIC. Review before installing: the kernel, bootloader and
# filesystem tools here assume an AMD CPU + btrfs + GRUB/EFI setup.
base                         # Minimal package set defining a basic Arch install
base-devel                   # Basic tools to build Arch packages
linux                        # The Linux kernel and modules
linux-headers                # Kernel headers -- required by DKMS modules
linux-firmware               # Firmware files for Linux
amd-ucode                    # Microcode for AMD CPUs -- swap for intel-ucode on Intel
grub                         # Bootloader
grub-btrfs                   # Include btrfs snapshots in GRUB boot options
os-prober                    # Detect other OSes on the drives (dual boot)
efibootmgr                   # Modify the EFI boot manager
btrfs-progs                  # Btrfs filesystem utilities
ntfs-3g                      # NTFS driver -- read Windows partitions
ntfsprogs                    # NTFS userspace utilities
zram-generator               # Systemd generator for zram swap
timeshift                    # System restore utility (btrfs snapshots)
smartmontools                # Monitor S.M.A.R.T. drive health
dmidecode                    # Read hardware info from the DMI table
reflector                    # Retrieve and filter the latest mirror list
rate-mirrors                 # Client-side mirror ranking
pacman-contrib               # Extra pacman tools (paccache, checkupdates)
pkgfile                      # Search for files in official repo packages
man-db                       # Read man pages

# --- NVIDIA ----------------------------------------------------------------
# ONLY if you have an NVIDIA GPU. On AMD/Intel delete this whole section and
# install mesa + vulkan-radeon (or vulkan-intel) instead.

lib32-nvidia-utils           # NVIDIA driver utilities, 32-bit (Steam needs this)

vulkan-tools                 # Vulkan utilities (vulkaninfo, vkcube)
lib32-vulkan-icd-loader      # Vulkan loader, 32-bit
