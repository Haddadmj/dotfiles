# Explicitly-installed AUR packages.
#
# PREREQUISITE: paru itself must already exist -- it's what installs this list,
# so it can't be in it. Bootstrap it from source first (this is also the step
# that has to happen before cloning this repo):
#
#   sudo pacman -S --needed base-devel git
#   git clone https://aur.archlinux.org/paru.git /tmp/paru
#   cd /tmp/paru && makepkg -si
#
# Then install (the sed strips these comments):
#   sed 's/#.*//' packages/aur.txt | paru -S --needed -
#
# Only the "Desktop" section matters for this config. Everything below it is
# personal software -- skim it and delete what you don't want rather than
# installing the lot.

# --- Desktop ---------------------------------------------------------------
# Needed by this config.
wlogout                      # Logout menu -- bound in keybinds.lua, themed by matugen
bibata-cursor-theme          # Cursor theme -- autostart.lua sets Bibata-Modern-Ice

hyprqt6engine                # Qt6 theme provider for Hyprland
noto-fonts-ar                # Noto Arabic -- drop if you don't need Arabic script
ttf-amiri                    # Amiri Arabic typeface -- drop if you don't need Arabic
